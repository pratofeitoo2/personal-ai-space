"""
GitHub Specialist Agent.

Manages git repositories, branches, worktrees, commits, and syncs.
Maintains bidirectional communication with all other agents to track work.

Registered commands (defined in agents.config.json):
  status, branch_list, branch_create, branch_delete, commit, log,
  pull, push, clone, init, worktree_add, worktree_list,
  repo_discover, repo_register, repo_list, sync_now,
  gh_repo_info, gh_pr_list, gh_ci_status
"""
import sys
from pathlib import Path
from datetime import datetime

sys.path.insert(0, str(Path(__file__).parent.parent))

from agents.base_agent import BaseAgent
from log_manager import audit


class GitHubAgent(BaseAgent):
    """Orchestrates git/GitHub operations and cross-agent tracking."""

    def __init__(self):
        super().__init__("github-agent")
        self._discovery = None      # GitHubDiscovery (wired in later from T7)
        self._git_ops = None        # GitOps instance (wired in later from T5)
        self._gh_api = None         # GitHubAPI instance (wired in later from T6)
        self._sync_engine = None    # SyncEngine instance (wired in later from T10)
        self._subscribed_events = {}  # event_type → handler method
        self._registered_repos_cache = []  # cached list of registered repos

    def initialize(self) -> bool:
        """Set up agent. Lazy-init helpers — they are wired from outside."""
        self.state = "ready"
        self.logger.info("GitHub Agent ready (helpers will be wired by T5-T10)")
        return True

    def shutdown(self) -> None:
        """Clean shutdown. Flush pending syncs."""
        if self._sync_engine:
            try:
                self._sync_engine.stop()
            except Exception as e:
                self.logger.warning(f"Sync engine stop error: {e}")
        self._subscribed_events.clear()
        self.state = "stopped"
        super().shutdown()

    # ── message routing ───────────────────────────────────────────────

    def process(self, message: dict) -> dict:
        """Route incoming message by payload.command to handler."""
        payload = message.get("payload", {})
        command = payload.get("command", "unknown")
        params = payload.get("parameters", {})

        handler = self._get_handler(command)
        if handler is None:
            return self._unknown(command)

        # Guardrail: check registered repo for operations that need it
        repo_path = params.get("repo_path")
        if repo_path and command not in ("repo_discover", "repo_register", "repo_list"):
            if not self._is_registered_repo(repo_path):
                return self._error(
                    f"Guardrail: repo not registered: {repo_path}. "
                    f"Use repo_discover or repo_register first."
                )

        try:
            return self._ok(handler(params))
        except Exception as e:
            self.logger.error(f"Command [{command}] failed: {e}", exc_info=True)
            return self._error(str(e))

    def _get_handler(self, command: str):
        """Map command name → handler method."""
        handlers = {
            "status": self._cmd_status,
            "branch_list": self._cmd_branch_list,
            "commit": self._cmd_commit,
            "log": self._cmd_log,
            "repo_list": self._cmd_repo_list,
            "sync_now": self._cmd_sync_now,
            "repo_discover": self._cmd_repo_discover,
            "repo_register": self._cmd_repo_register,
            # T5-T10 will wire more handlers
        }
        return handlers.get(command)

    # ── base command handlers ─────────────────────────────────────────

    def _cmd_status(self, params: dict) -> dict:
        """Return git status for a repo. Requires repo_path param."""
        repo_path = params.get("repo_path")
        if not repo_path:
            return {"message": "GitHub Agent ready. Use repo_path to check specific repo status.",
                    "repos_tracked": len(self._registered_repos_cache)}
        # Real implementation uses GitOps (T5) — for now return placeholder
        return {
            "repo_path": repo_path,
            "note": "Git operations not yet wired — requires GitOps module (T5)",
        }

    def _cmd_branch_list(self, params: dict) -> dict:
        """List branches for a repo."""
        repo_path = params.get("repo_path")
        if not repo_path:
            return {"branches": []}
        return {"repo_path": repo_path,
                "note": "Branch operations not yet wired (T5)"}

    def _cmd_commit(self, params: dict) -> dict:
        """Commit staged changes."""
        return {"note": "Commit operations not yet wired (T5)"}

    def _cmd_log(self, params: dict) -> dict:
        """Show recent commit log."""
        return {"commits": [],
                "note": "Log operations not yet wired (T5)"}

    def _cmd_repo_list(self, params: dict) -> dict:
        """List all registered repos."""
        return {
            "repos": self._registered_repos_cache,
            "count": len(self._registered_repos_cache),
        }

    def _cmd_sync_now(self, params: dict) -> dict:
        """Trigger immediate sync."""
        return {"synced": False,
                "note": "Sync engine not yet wired (T10)"}

    def _cmd_repo_discover(self, params: dict) -> dict:
        """Discover git repos in given paths."""
        paths = params.get("paths", [])
        return {"discovered": [],
                "note": "Discovery not yet wired (T7)"}

    def _cmd_repo_register(self, params: dict) -> dict:
        """Register a repo path for tracking."""
        return {"registered": False,
                "note": "Discovery not yet wired (T7)"}

    # ── guardrails ────────────────────────────────────────────────────

    def _is_registered_repo(self, repo_path: str) -> bool:
        """Check if a repo path is registered (for guardrail enforcement)."""
        if not repo_path:
            return False
        # When discovery (T7) is wired, this checks git.db
        # Until then, allow the main project repo
        if "Personal_AI_powerhouse" in repo_path:
            return True
        return any(
            r["path"] == repo_path
            for r in self._registered_repos_cache
        )

    # ── cross-agent integration (will be wired by T8, T9) ─────────────

    def set_observer(self, observer):
        """Wire observer (called by engine). Overrides BaseAgent."""
        super().set_observer(observer)
        # T9 will integrate observer here
        self.logger.debug("Observer wired to GitHub agent")

    def subscribe_to_event(self, event_type: str, handler):
        """Register interest in a cross-agent event type."""
        self._subscribed_events[event_type] = handler

    def _handle_event(self, event_type: str, data: dict) -> dict:
        """Route an incoming cross-agent event to its handler."""
        handler = self._subscribed_events.get(event_type)
        if handler:
            return handler(data)
        self.logger.debug(f"No handler for event: {event_type}")
        return {"handled": False, "event_type": event_type}

    # ── helper wiring methods (called by T5-T10) ──────────────────────

    def wire_git_ops(self, git_ops):
        """Wire the GitOps helper (called by T5)."""
        self._git_ops = git_ops

    def wire_gh_api(self, gh_api):
        """Wire the GitHub API helper (called by T6)."""
        self._gh_api = gh_api

    def wire_discovery(self, discovery):
        """Wire the repo discovery module (called by T7)."""
        self._discovery = discovery
        if discovery:
            self._registered_repos_cache = discovery.get_registered_repos()

    def wire_sync_engine(self, sync_engine):
        """Wire the sync engine (called by T10)."""
        self._sync_engine = sync_engine
